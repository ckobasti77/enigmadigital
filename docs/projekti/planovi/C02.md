### 2.1 Tipovi i jezgro
```ts
import { ConvexError, v } from "convex/values";
import { getAuthUserId } from "@convex-dev/auth/server";
import {
  customAction,
  customMutation,
  customQuery,
} from "convex-helpers/server/customFunctions";
import {
  action,
  mutation,
  query,
  type ActionCtx,
  type MutationCtx,
  type QueryCtx,
} from "../_generated/server";
import { internal } from "../_generated/api";
import type { Id } from "../_generated/dataModel";
import { imaUlogu, ulogaValidator, type Uloga } from "./uloge";

export type Projekat = {
  userId: Id<"users">;
  workspaceId: Id<"workspaces">;
  uloga: Uloga;
};
export type ProjekatQueryCtx = QueryCtx & { projekat: Projekat };
export type ProjekatMutationCtx = MutationCtx & { projekat: Projekat };
export type ProjekatActionCtx = ActionCtx & { projekat: Projekat };

export const PORUKE = {
  nijePrijavljen: { code: "unauthorized", message: "Niste prijavljeni." },
  nemaPristup: { code: "forbidden", message: "Nemaš pristup ovom projektu." },
  urednik: { code: "forbidden", message: "Ovu radnju može da izvede urednik ili vlasnik projekta." },
  owner: { code: "forbidden", message: "Ovu radnju može da izvede samo vlasnik projekta." },
  platforma: { code: "forbidden", message: "Ovu radnju može da izvede samo vlasnik platforme." },
} as const;

export async function prijavljenKorisnik(ctx: QueryCtx | ActionCtx): Promise<Id<"users">> {
  const userId = await getAuthUserId(ctx);
  if (userId === null) throw new ConvexError(PORUKE.nijePrijavljen);
  return userId;
}

/** Članstvo korisnika u TAČNO ovom projektu, sa minimalnom ulogom. Inače baca. */
export async function clanstvoKorisnika(
  ctx: QueryCtx,
  userId: Id<"users">,
  workspaceId: Id<"workspaces">,
  minimalna: Uloga,
): Promise<Projekat> {
  const clan = await ctx.db
    .query("members")
    .withIndex("by_user_workspace", (q) =>
      q.eq("userId", userId).eq("workspaceId", workspaceId),
    )
    .first(); // D14
  if (clan === null) throw new ConvexError(PORUKE.nemaPristup);
  if (!imaUlogu(clan.role, minimalna)) {
    throw new ConvexError(minimalna === "owner" ? PORUKE.owner : PORUKE.urednik);
  }
  return { userId, workspaceId, uloga: clan.role };
}

/** Za mesta van omotača (unakrsno kopiranje u C17, testovi). */
export async function requireProjekat(
  ctx: QueryCtx,
  workspaceId: Id<"workspaces">,
  minimalna: Uloga = "client_viewer",
): Promise<Projekat> {
  return clanstvoKorisnika(ctx, await prijavljenKorisnik(ctx), workspaceId, minimalna);
}

/** P3: `owner` u bar jednom internom projektu (vrsta bez vrednosti = interni). */
export async function jeVlasnikPlatforme(ctx: QueryCtx, userId: Id<"users">): Promise<boolean> {
  const clanstva = await ctx.db
    .query("members")
    .withIndex("by_user", (q) => q.eq("userId", userId))
    .take(100);
  for (const c of clanstva) {
    if (c.role !== "owner") continue;
    const ws = await ctx.db.get(c.workspaceId);
    if (ws !== null && (ws.vrsta ?? "interni") === "interni") return true;
  }
  return false;
}
```

  ...GLOBALNE_PUTANJE, ...HUB_PUTANJE, ...OSTALI, ...EKRANI,
]);

/** 2–40 znakova: mala slova, cifre, crtica; bez vodeće ili prateće crtice i bez „--“. */
export const SLUG_OBRAZAC = /^[a-z0-9](?:[a-z0-9]|-(?=[a-z0-9])){1,39}$/;

export function proveriSlugProjekta(slug: string): { ok: true } | { ok: false; razlog: string } {
  if (!SLUG_OBRAZAC.test(slug)) {
    return { ok: false, razlog: "Adresa sme da ima mala slova, cifre i crtice (2–40 znakova), bez crtice na početku i kraju." };
  }
  if (REZERVISANI_SLUGOVI.has(slug)) {
    return { ok: false, razlog: "Ova adresa je rezervisana. Izaberi drugu." };
  }
  return { ok: true };
}
export function jeIspravanSlug(slug: string): boolean {
  return proveriSlugProjekta(slug).ok;
}

export function normalizujHost(host: string | null | undefined): string | null {
  if (!host) return null;
  let h = host.trim().toLowerCase();
  if (h.startsWith("[")) return null;                      // IPv6 literal nije panel
  const dvotacka = h.indexOf(":");
  if (dvotacka !== -1) h = h.slice(0, dvotacka);           // bez porta
  if (h.endsWith(".")) h = h.slice(0, -1);
  if (h.length === 0 || h.length > 253 || !/^[a-z0-9.-]+$/.test(h)) return null;
  return h;
}
export function jeLokalniHost(h: string): boolean {
  return h === "localhost" || h === "127.0.0.1" || h.endsWith(".localhost");
}

## 4. Recept za prelazak modula (C04–C07)

### 4.1 Backend, za svaku javnu funkciju iz mape (§5.1)
1. Registraciju zameni omotačem iz mape: `query(` → `projekatQuery(`, `mutation(` → `projekatMutation(`, `action(` → `projekatAction(`. `uloga` piši samo kad je drugačija od podrazumevane (D4). Uvoz ide iz `./lib/projekat`.
2. Iz sopstvenih `args` funkcije **ukloni `workspaceId`** (omotač ga dodaje kao obavezan, P2). Opcioni `workspaceId` (googleAdsStore, metaAds, rulesStore, metaCapi, adActions, attribution) takođe nestaje.
3. Obriši `requireMembership`, `requireOwner`, `requireMembershipIn`, `requireWorkspaceAccess`, `requireActionMembership` i ručno poređenje `membership.workspaceId !== args.workspaceId`. Prostor je `ctx.projekat.workspaceId` (ili `args.workspaceId`, D3), korisnik je `ctx.projekat.userId`.
4. **Zadrži ili dodaj** proveru dokumenta iz argumenta: `doc === null || doc.workspaceId !== workspaceId` → `not_found` ili prazan rezultat (isto ponašanje kao do sada).
5. Akcije: interna funkcija koja je danas zvala `requireMembership(ctx)` preko prenetog identiteta (npr. `igCommentsStore.loadModerationContext`, `orSend.loadInboxSendContext`) dobija argumente `workspaceId` (i `userId` kad ga koristi) i više ne proverava članstvo (D6). Akcija ih šalje iz `ctx.projekat`.
6. Pomoćne funkcije „prvo članstvo“ (§5.2): pozivaoci prelaze na `ctx.projekat`. Pomoćna funkcija se briše u koraku u kome joj nestane poslednji pozivalac.
7. Članstvo ciljnog korisnika (npr. `assignLead.ownerUserId`) se čita preko `members.by_user_workspace`, ne `by_user`.
8. Poruke koje korisnik vidi u **dirnutim** funkcijama: „radni prostor“ → „projekat“ (P1). Ostali kod se ne dira.
9. Funkcije sa `// javno:` ostaju sirove registracije (OAuth povratak, registracija, `postsPublic`, `getDeletionStatus`, `razresiDomen`).
10. Interne funkcije, HTTP akcije i cron se ne prebacuju. Prostor i dalje nose eksplicitno ili ga nalaze po ID-ju naloga.
11. Server koji zove prebačenu javnu funkciju mora da pošalje `workspaceId` ili da pređe na internu funkciju. Jedino takvo mesto je `orSend.ts:1390`: C04 mu dodaje `workspaceId: kontekst.workspaceId` (jedan red), a ostatak `orSend` je C05.
12. `convex/pristup.test.ts` i `convex/threadsBrisanje.test.ts`: kad se potpis prebačene funkcije promeni, **menja se samo poziv** (doda se ili ukloni `workspaceId`). Tvrdnje (`unauthorized`/`forbidden`/prazno) ostaju (pravila runa).

### 4.2 Front, za svaki poziv prebačene funkcije
1. `useQuery(api.x.y, { workspaceId, ...ostalo })` ili `useQuery(api.x.y, args)` → `useProjekatQuery(api.x.y, ostalo)`. `"skip"` ostaje `"skip"`.
2. `useMutation(api.x.y)` → `useProjekatMutation(api.x.y)`. Na mestu poziva samo ukloni `workspaceId` iz objekta.
3. `useAction(api.x.y)` → `useProjekatAction(api.x.y)`.
4. `useWorkspace()` u dirnutom fajlu → `useProjekat()`:
   - `workspace.id` → `workspaceId`;
   - `workspace.slug` → `slug`;
   - `workspace.name` → `ime`;
   - `user.id` / `user.email` / `user.name` → `korisnik.id` / `korisnik.email` / `korisnik.ime`;
   - `role` → `uloga`;
   - `isLoading` → `!spreman`.
5. Poziv nikad ne šalje `workspaceId` ručno i nikad ne bira projekat.
6. Ako komponenta zove funkcije više grupa, korak menja **samo pozive svoje grupe**. TypeScript ne dozvoljava `useProjekatQuery` nad neprebačenom funkcijom.
7. Fiksture (`app/dev-ux/*`) su ključne po imenu funkcije i ne menjaju se. Resolver koji čita `args` prima i `workspaceId`; to sme da ignoriše.

### 6.3 Hookovi (tačni tipovi)
```ts
import type { FunctionArgs, FunctionReference, FunctionReturnType } from "convex/server";

type SaProjektom = { workspaceId: Id<"workspaces"> };
type BezProjekta<F extends FunctionReference<"query" | "mutation" | "action">> =
  Omit<FunctionArgs<F>, "workspaceId">;
type ArgsUpita<Q extends FunctionReference<"query">> =
  {} extends BezProjekta<Q> ? [args?: BezProjekta<Q> | "skip"] : [args: BezProjekta<Q> | "skip"];
type ArgsPoziva<F extends FunctionReference<"mutation" | "action">> =
  {} extends BezProjekta<F> ? [args?: BezProjekta<F>] : [args: BezProjekta<F>];

export function useProjekatQuery<Q extends FunctionReference<"query", "public", SaProjektom>>(
  upit: Q,
  ...args: ArgsUpita<Q>
): FunctionReturnType<Q> | undefined {
  const { workspaceId } = useProjekat();
  const a = args[0];
  const pun = a === "skip" || workspaceId === null
    ? "skip"
    : ({ ...(a ?? {}), workspaceId } as FunctionArgs<Q>);   // workspaceId uvek poslednji: pregazi sve
  return useQuery(upit as FunctionReference<"query">, pun);  // jedini `as`, samo ovde
}

export function useProjekatMutation<M extends FunctionReference<"mutation", "public", SaProjektom>>(m: M) {
  const { workspaceId } = useProjekat();
  const pozovi = useMutation(m);
  return useCallback(
    (...args: ArgsPoziva<M>): Promise<FunctionReturnType<M>> =>
      workspaceId === null
        ? Promise.reject(new Error("Projekat još nije učitan."))
        : pozovi({ ...(args[0] ?? {}), workspaceId } as FunctionArgs<M>),
    [pozovi, workspaceId],
  );
}
// useProjekatAction: isto, sa useAction.
```
Kako `workspaceId` ispada iz argumenata:
- ograničenje `Q extends FunctionReference<…, SaProjektom>` odbija funkciju bez `workspaceId`;
- `Omit<…, "workspaceId">` ga uklanja iz argumenata, pa doslovan objekat sa `workspaceId` pada na proveri viška svojstava;
- u runtime-u se `workspaceId` upisuje poslednji, pa ga nijedan prosleđen objekat ne menja.

**`odluciRutu`** (pravila iz §6, ovim redom):
1. Globalna putanja → `dalje`.
2. Panel (`panelSlug = S`):
   - `/S` ili `/S/…` → `redirect` na putanju bez `/S` (prazno → `/`), uz `search`;
   - inače → `rewrite` `/S` + putanja (za `/` → `/S`), uz `search`.
3. Hub:
   - `/` → `dalje`;
   - prvi segment ∈ `EKRANI` → `redirect`: kolačić prolazi `jeIspravanSlug` → `/<kolačić>` + putanja + `search`, inače `/` + `search` (čuva `ig_code`);
   - prvi segment ∈ `HUB_PUTANJE` → `dalje`;
   - ostalo → `dalje` (layout `[projekat]` proverava oblik, provider članstvo).
4. `localhost`, `127.0.0.1` i `*.localhost` su hub bez upita, a greška upita je hub. To rešava razrešavač (§7.5), pa `odluciRutu` dobija `panelSlug: null`.

### 7.3 `proxy.ts`
Postojeći komentari i `config.matcher` ostaju **doslovno**. Telo:
```ts
import { fetchQuery } from "convex/nextjs";
import { api } from "@/convex/_generated/api";
import { resolveServerConvexUrl } from "@/lib/server-convex-url";
import { KOLACIC_PROJEKAT, jeLokalniHost, normalizujHost, odluciRutu } from "@/lib/rutiranje";
import { napraviRazresivac } from "@/lib/panel-domen";

const razresiPanel = napraviRazresivac({
  dohvati: async (host) => {
    const url = resolveServerConvexUrl();
    if (!url.ok) throw new Error(url.logDetail);
    const r = await fetchQuery(api.projekti.razresiDomen, { host }, { url: url.url });
    return r?.slug ?? null;
  },
});

export default convexAuthNextjsMiddleware(async (request, { convexAuth }) => {
  // 1. Razvojni prikaz: blok bez izmene, ostaje PRVI.
  // 2. Režim:
  const host = normalizujHost(request.headers.get("host"));
  const panelSlug = host === null || jeLokalniHost(host) ? null : await razresiPanel(host);
  const zaglavlja = new Headers(request.headers);           // već nosi kolačiće koje je Convex Auth osvežio
  zaglavlja.set("x-panel-rezim", panelSlug ? "panel" : "hub");
  if (panelSlug) zaglavlja.set("x-panel-projekat", panelSlug);
  else zaglavlja.delete("x-panel-projekat");                // nikad ne prenosi vrednost koju je poslao klijent
  // 3. Prijava: postojeći redosled i kod bez izmene (/login + authed → "/"; neprijavljen → /login sa ig_code).
  // 4. Rutiranje (samo prijavljeni ili /login):
  const odluka = odluciRutu({
    pathname: request.nextUrl.pathname,
    search: request.nextUrl.search,
    panelSlug,
    kolacicProjekat: request.cookies.get(KOLACIC_PROJEKAT)?.value,
  });
  if (odluka.tip === "redirect") return nextjsMiddlewareRedirect(request, odluka.putanja);
  if (odluka.tip === "rewrite") {
    return NextResponse.rewrite(new URL(odluka.putanja, request.url), { request: { headers: zaglavlja } });
  }
  return NextResponse.next({ request: { headers: zaglavlja } });
});
```

- `/login` za prijavljene;
- `?code=` koji presreće Convex Auth;
- izuzeci u oba matcher-a;
- `/r/:slug` u `next.config.ts`;
- `/api/auth*` (Convex Auth ga obrađuje pre handler-a).

`lib/boja-projekta.ts` (čista funkcija):
```ts
export function izvediBojuProjekta(hex: string | null): { boja: string; tekst: string } | null;
```
- `null` ili neispravan hex → `null` (važi podrazumevani token).
- `tekst` je tamni (`--background` #070d19) ili svetli (`--foreground` #f3f7ff), onaj sa većim kontrastom.

- Boja mora imati bar 3:1 prema `--background` (traka i prsten). Ako nema, posvetli se.
- Test čita `app/globals.css` i proverava da su dve heks vrednosti iste kao tokeni, da izvor istine ostane CSS.

`ProjekatProvider` ili koren `AppShell` postavlja `style={{ "--projekat": boja, "--projekat-tekst": tekst }}`. Vrednost dolazi iz podataka, nije zakucan hex.
